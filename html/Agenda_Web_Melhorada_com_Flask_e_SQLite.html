Do protótipo ao sistema: a evolução do projeto

A agenda anterior guardava contatos em uma lista Python. A nova versão resolve as limitações dela.

Antes: agenda básicaAgora: agenda web melhorada
ArmazenamentoLista Python em memória (volátil)→Banco de dados SQLite em instance/agenda.db
UsuáriosAnônimo e único: todos veem tudo→Vários usuários com login; cada um vê só os seus contatos
Ciclo de vidaSome ao reiniciar o servidor→Dados persistem; o login é mantido entre páginas pela sessão
InterfaceUma página com formulário e lista→Menu, dashboard, formulários e busca com Bootstrap
ArquiteturaScript linear único→MVC simplificado: dados, controle e visão separados

O objetivo é entender o projeto não como trechos de código soltos, mas como um sistema em que cada peça depende das outras.

O esqueleto relacional: dados separados por usuário

Duas tabelas ligadas por uma chave estrangeira.

users

contacts
🔑id INTEGER
FKuser_id → users.id

email TEXT
phone TEXT

Um usuário tem muitos contatos; cada contato pertence a exatamente um usuário.

FK

user_id guarda o id do dono. É a chave estrangeira que liga as tabelas.

UQ

email UNIQUE em users: o banco impede dois cadastros com o mesmo e-mail.

🔒

A senha nunca é guardada, só o seu hash (Parte 3).

A regra de ouro: toda consulta a contacts inclui WHERE user_id = ? com o id da sessão. É isso que garante que cada usuário veja apenas os seus dados.

def init_db():
    """Cria as tabelas usando comandos SQL simples."""
    db = get_db()
    db.execute(
        """
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            email TEXT NOT NULL UNIQUE,
            password_hash TEXT NOT NULL,
            created_at TEXT NOT NULL
        )
        """
    )
    db.execute(
        """
        CREATE TABLE IF NOT EXISTS contacts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER NOT NULL,
            name TEXT NOT NULL,
            email TEXT,
            phone TEXT,
            notes TEXT,
            created_at TEXT NOT NULL,
            FOREIGN KEY (user_id) REFERENCES users (id)
        )
        """
    )
    db.commit()

BASE_DIR = os.path.abspath(os.path.dirname(__file__))
DATABASE = os.path.join(BASE_DIR, "instance", "agenda.db")

app = Flask(__name__)
app.config["SECRET_KEY"] = os.environ.get("SECRET_KEY", "dev-secret-key-troque-em-producao")

def get_db():
    """Abre uma conexão SQLite por requisição."""
    if "db" not in g:
        os.makedirs(os.path.dirname(app.config["DATABASE"]), exist_ok=True)
        g.db = sqlite3.connect(app.config["DATABASE"])
        g.db.row_factory = sqlite3.Row
    return g.db


@app.teardown_appcontext
def close_db(error=None):
    db = g.pop("db", None)
    if db is not None:
        db.close()

email = request.form["email"]
# Um atacante digita no campo e-mail:   ' OR '1'='1

# ❌ NUNCA: montar o SQL concatenando texto
sql = f"SELECT * FROM users WHERE email = '{email}'"
# O banco recebe:
#   SELECT * FROM users WHERE email = '' OR '1'='1'
# '1'='1' é sempre verdadeiro → devolve TODOS os usuários

# ✅ SEMPRE: marcadores ? e valores separados (como no projeto)
db.execute("SELECT * FROM users WHERE email = ?", (email,))
# O texto ' OR '1'='1 é tratado como um e-mail estranho,
# não como comando. Resultado: nenhuma linha.

@app.route("/register", methods=["GET", "POST"])
def register():
    if request.method == "POST":
        name = request.form.get("name", "").strip()
        email = request.form.get("email", "").strip().lower()
        password = request.form.get("password", "")
        confirm_password = request.form.get("confirm_password", "")

        if not name or not email or not password:
            flash("Preencha todos os campos obrigatórios.", "danger")
        elif len(password) < 6:
            flash("A senha deve ter pelo menos 6 caracteres.", "danger")
        elif password != confirm_password:
            flash("A confirmação de senha não confere.", "danger")
        else:
            try:
                get_db().execute(
                    "INSERT INTO users (name, email, password_hash, created_at) VALUES (?, ?, ?, ?)",
                    (name, email, generate_password_hash(password), datetime.now().isoformat(timespec="seconds")),
                )
                get_db().commit()
                flash("Usuário cadastrado com sucesso. Agora faça login.", "success")
                return redirect(url_for("login"))
            except sqlite3.IntegrityError:
                flash("Este e-mail já está cadastrado.", "danger")

    return render_template("register.html")

@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        email = request.form.get("email", "").strip().lower()
        password = request.form.get("password", "")

        user = get_db().execute(
            "SELECT * FROM users WHERE email = ?", (email,)
        ).fetchone()

        if user is None or not check_password_hash(user["password_hash"], password):
            flash("E-mail ou senha inválidos.", "danger")
        else:
            session.clear()
            session["user_id"] = user["id"]
            session["user_name"] = user["name"]
            flash(f"Bem-vindo(a), {user['name']}!", "success")
            return redirect(url_for("dashboard"))

    return render_template("login.html")

from functools import wraps

def anunciar(funcao):
    @wraps(funcao)
    def envoltorio(*args, **kwargs):
        print(f"Antes de {funcao.__name__}")
        resultado = funcao(*args, **kwargs)
        print("Depois")
        return resultado
    return envoltorio

@anunciar
def somar(a, b):
    return a + b

print(somar(2, 3))

Um ORM gera o SQL (sempre parametrizado), facilita trocar SQLite por PostgreSQL e versionar o esquema com migrações.
